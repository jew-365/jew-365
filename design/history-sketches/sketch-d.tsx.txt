function greeting(profileName: string): string {
  const first = profileName.trim().split(/\s+/)[0];
  return first ? `שלום, ${first}` : 'שלום וברוכים הבאים';
}

function PercentRing({ done, total, size }: { done: number; total: number; size: number }) {
  const { colors } = useTheme();
  const percent = Math.round((done / total) * 100);
  const stroke = Math.round(size * 0.055);
  const center = size / 2;
  const r = (size - stroke) / 2;
  const circumference = 2 * Math.PI * r;
  return (
    <View accessible accessibilityLabel={`${percent}%, ${done} מתוך ${total}`} style={{ width: size, height: size }}>
      <Svg width={size} height={size} style={StyleSheet.absoluteFill}>
        <Circle cx={center} cy={center} r={r} fill={colors.surface} fillOpacity={DISC_OPACITY} />
        <Circle cx={center} cy={center} r={r} fill="none" stroke={colors.surface2} strokeWidth={stroke} />
        <Circle
          cx={center}
          cy={center}
          r={r}
          fill="none"
          stroke={colors.gold}
          strokeWidth={stroke}
          strokeLinecap="round"
          strokeDasharray={`${circumference} ${circumference}`}
          strokeDashoffset={circumference * (1 - done / total)}
          transform={`rotate(-90 ${center} ${center})`}
        />
      </Svg>
      <View style={styles.ringCenter}>
        <Text
          style={{
            fontFamily: fontFamilies.heebo.black,
            fontSize: Math.round(size * 0.27),
            lineHeight: Math.round(size * 0.31),
            color: colors.text,
          }}
        >
          {percent}%
        </Text>
        <Text style={[typography.caption, { color: colors.textSub }]}>
          {done} מתוך {total}
        </Text>
      </View>
    </View>
  );
}

function PeriodHead({ done, total, title, caption }: { done: number; total: number; title: string; caption: string }) {
  const { colors } = useTheme();
  return (
    <View style={styles.periodHead}>

      <Text style={[typography.title, styles.center, { color: colors.text }]}>{title}</Text>
      <Text style={[typography.small, styles.center, styles.explain, { color: colors.textSub }]}>{caption}</Text>
    </View>
  );
}
