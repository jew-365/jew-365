const pct = (done: number, total: number) => Math.round((done / total) * 100);

function PercentRing({ done, total, size }: { done: number; total: number; size: number }) {
  const { colors } = useTheme();
  const percent = pct(done, total);
  const stroke = Math.round(size * 0.075);
  const center = size / 2;
  const r = (size - stroke) / 2;
  const circumference = 2 * Math.PI * r;
  return (
    <View accessible accessibilityLabel={`${percent}%, ${done} מתוך ${total}`} style={{ width: size, height: size }}>
      <Svg width={size} height={size} style={StyleSheet.absoluteFill}>
        <Circle cx={center} cy={center} r={r} fill={colors.surface} fillOpacity={DISC_OPACITY} />
        <Circle cx={center} cy={center} r={r} fill="none" stroke={colors.surface2} strokeWidth={stroke} />
        <Circle
          cx={center}
          cy={center}
          r={r}
          fill="none"
          stroke={colors.gold}
          strokeWidth={stroke}
          strokeLinecap="round"
          strokeDasharray={`${circumference} ${circumference}`}
          strokeDashoffset={circumference * (1 - done / total)}
          transform={`rotate(-90 ${center} ${center})`}
        />
      </Svg>
      <View style={styles.ringCenter}>
        <Text
          style={{
            fontFamily: fontFamilies.heebo.black,
            fontSize: Math.round(size * 0.26),
            lineHeight: Math.round(size * 0.3),
            color: colors.text,
          }}
        >
          {percent}%
        </Text>
        <Text style={[typography.small, { color: colors.textSub }]}>
          {done}/{total}
        </Text>
      </View>
    </View>
  );
}

function WeeksChart() {
  const { colors } = useTheme();
  const finished = WEEKS.filter((w) => !w.current);
  const best = finished.reduce((a, b) => (pct(b.done, b.total) > pct(a.done, a.total) ? b : a));
  return (
    <Card padded>
      <View
        accessible
        accessibilityLabel={WEEKS.map((w) => `${w.name} ${pct(w.done, w.total)}%`).join(', ')}
        style={styles.weekChart}
      >
        <View style={[styles.gridLine, { top: 0, backgroundColor: colors.border }]} />
        <View style={[styles.gridLine, { top: WEEK_CHART_HEIGHT / 2, backgroundColor: colors.border }]} />
        <View style={[styles.gridLine, { bottom: 0, backgroundColor: colors.border }]} />
        {WEEKS.map((w) => {
          const value = pct(w.done, w.total);
          const labelled = w.current || w === best;
          return (
            <View key={w.name} style={styles.weekColumn}>
              {labelled ? (
                <Text style={[typography.micro, { color: w.current ? colors.goldText : colors.textSub }]}>{value}%</Text>
              ) : null}
              <View style={[styles.weekBar, { height: (WEEK_CHART_HEIGHT * value) / 100, backgroundColor: colors.surface }]}>
              <View
                style={[
                  styles.weekBar,
                  styles.flex,
                  { height: (WEEK_CHART_HEIGHT * value) / 100 },
                  w.current
                    ? { backgroundColor: colors.goldLight, borderWidth: 1.5, borderColor: colors.gold, borderStyle: 'dashed' }
                    : { backgroundColor: colors.gold, opacity: w === best ? 1 : MUTED_BAR_OPACITY },
                ]}
              />
              </View>
            </View>
          );
        })}
        <Text style={[typography.micro, styles.axisTop, { color: colors.textSub }]}>100%</Text>
        <Text style={[typography.micro, styles.axisMid, { color: colors.textSub }]}>50%</Text>
      </View>
      <View style={styles.axisRow}>
        <Text style={[typography.small, { color: colors.textSub }]}>שופטים</Text>
        <Text style={[typography.small, { color: colors.goldText }]}>השבוע, עד עכשיו</Text>
      </View>
    </Card>
  );
}

function MitzvahTrends() {
  const { colors } = useTheme();
  const sorted = [...MITZVOT].sort((a, b) => b.percent - b.previous - (a.percent - a.previous));
  return (
    <Card divided>
      {sorted.map((m) => {
        const delta = m.percent - m.previous;
        const up = delta > 0;
        return (
          <View key={m.id} style={styles.statRow}>
            <View style={styles.statTop}>
              <Text style={[typography.bodyBold, styles.flex, { color: colors.text }]} numberOfLines={1}>
                {m.name}
              </Text>
              <View style={[styles.chip, { backgroundColor: up ? colors.goldLight : colors.surface2 }]}>
                {delta !== 0 ? (
                  <MitzvahIcon
                    name={up ? 'chevronUp' : 'chevronDown'}
                    size={DELTA_ICON}
                    color={up ? colors.goldText : colors.textSub}
                    strokeWidth={2.5}
                  />
                ) : null}
                <Text style={[typography.small, { color: up ? colors.goldText : colors.textSub }]}>
                  {delta === 0 ? 'ללא שינוי' : `${Math.abs(delta)} נק׳`}
                </Text>
              </View>
              <Text style={[typography.captionBold, styles.percent, { color: colors.goldText }]}>{m.percent}%</Text>
            </View>
            <View>
              <Bar percent={m.percent} />
              <View style={styles.tickLayer} pointerEvents="none">
                <View style={{ width: `${m.previous}%` }} />
                <View style={[styles.previousTick, { backgroundColor: colors.text }]} />
              </View>
            </View>
          </View>
        );
      })}
    </Card>
  );
}
