function greeting(profileName: string): string {
  const first = profileName.trim().split(/\s+/)[0];
  return first ? `שלום, ${first}` : 'שלום וברוכים הבאים';
}

if (typeof window !== "undefined" && window.location) {
  const theme = new URLSearchParams(window.location.search).get("theme");
  useUserStore.setState({
    isOnboarded: true,
    language: "he",
    ...(theme && (THEME_NAMES as readonly string[]).includes(theme) ? { theme: theme as ThemeName } : {}),
  });
}

export default function SketchHistoryB() {
  const { colors } = useTheme();

  const numbers = [
    { value: `${STREAK}`, label: 'ימים ברצף', explain: 'ימים רצופים עם סימון אחד לפחות' },
    { value: `${BEST}`, label: 'השיא שלך', explain: 'הרצף הארוך ביותר אי פעם' },

  return (
    <SafeAreaView style={[styles.safe, { backgroundColor: colors.bg }]} edges={['top']}>

                  <Text style={[styles.number, { color: index === 0 ? colors.goldText : colors.text }]}>{n.value}</Text>
                  <Text style={[typography.captionBold, styles.center, { color: colors.text }]}>{n.label}</Text>

              </React.Fragment>
            ))}
          </View>
          <View style={[styles.hDivider, { backgroundColor: colors.border }]} />

            <Text style={[typography.captionBold, { color: colors.goldText }]}>עוד 18</Text>

          <View style={styles.dots}>

                  <View
                    key={col}
                    style={[

                      {

                      row * 10 + col === DAYS.length - 1 && { borderWidth: 1.5, borderColor: colors.goldText },
                    ]}
                  />

              </View>
            ))}
          </View>
          <Text style={[typography.small, { color: colors.textSub }]}>

          </Text>
        </Card>

        <SectionLabel text="ההישגים שלך" count={EARNED.length} />
        <View style={styles.earnedRow}>

            <Card key={badge.id} padded style={styles.earned}>

                <Text style={[typography.captionBold, styles.center, { color: colors.text }]}>{badge.title}</Text>

              </View>

          ))}
        </View>

                </View>
              }

          ))}
        </Card>
